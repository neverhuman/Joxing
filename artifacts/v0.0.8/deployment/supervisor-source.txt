"""Owned, bounded v8 validation -> registered run -> audit -> report wrapper."""
import datetime,json,os,pathlib,signal,subprocess,time,traceback,shutil
root=pathlib.Path('/home/ubuntu/joxing/v008-20261026')
os.chdir(root)
out=root/'runs/v0.0.8';out.mkdir(parents=True,exist_ok=True)
status={'pid':os.getpid(),'cwd':str(root),'started_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'source_commit':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),'status':'running','stages':[],'simulation_workers_cap':2,'threads':1}
def save():
 p=out/'supervisor.json';t=p.with_suffix('.tmp');t.write_text(json.dumps(status,indent=2)+'\n');t.replace(p)
def run(name,command,timeout):
 row={'stage':name,'command':command,'status':'running'};status['stages'].append(row);save();start=time.monotonic()
 with (out/(name+'.log')).open('w') as log:
  process=subprocess.Popen(command,stdout=log,stderr=subprocess.STDOUT,start_new_session=True)
  row['pid']=process.pid;save()
  try:rc=process.wait(timeout=timeout)
  except subprocess.TimeoutExpired:
   os.killpg(process.pid,signal.SIGTERM)
   try:process.wait(timeout=10)
   except subprocess.TimeoutExpired:os.killpg(process.pid,signal.SIGKILL);process.wait()
   row.update(status='timeout',wall_s=time.monotonic()-start,returncode=process.returncode);save();raise
 row.update(status='completed' if rc==0 else 'failed',wall_s=time.monotonic()-start,returncode=rc);save()
 if rc:raise RuntimeError(name+' returned '+str(rc))
try:
 save()
 run('sync',['uv','sync','--project','experiments/v0.0.8/runtime','--locked'],300)
 run('tests',['uv','run','--project','experiments/v0.0.8/runtime','--locked','pytest','-q'],600)
 owned=[]
 for entry in pathlib.Path('/proc').iterdir():
  if not entry.name.isdigit() or int(entry.name)==os.getpid():continue
  try:
   cwd=os.readlink(entry/'cwd');args=(entry/'cmdline').read_bytes().replace(b'\0',b' ').decode(errors='replace')
   if '/joxing/' in cwd and ('python' in args or 'uv run' in args):owned.append({'pid':int(entry.name),'cwd':cwd,'command':args})
  except (FileNotFoundError,PermissionError,ProcessLookupError):pass
 memory={line.split(':')[0]:line.split(':')[1].strip() for line in pathlib.Path('/proc/meminfo').read_text().splitlines()}
 capacity={'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'load':os.getloadavg(),'logical_cpus':os.cpu_count(),'available_memory_bytes':int(memory['MemAvailable'].split()[0])*1024,'free_disk_bytes':shutil.disk_usage(root).free,'other_owned_jobs':owned}
 (out/'capacity-before-run.json').write_text(json.dumps(capacity,indent=2)+'\n')
 if owned or capacity['available_memory_bytes']<8*2**30 or capacity['free_disk_bytes']<20*2**30 or capacity['load'][0]+2>capacity['logical_cpus']:raise RuntimeError('Fresh shared-host capacity gate')
 formal_start=time.monotonic()
 def remaining():return max(1,900-(time.monotonic()-formal_start))
 run('physical',['uv','run','--project','experiments/v0.0.8/runtime','--locked','python','scripts/run_physical_motor.py','--protocol','experiments/v0.0.8/protocol.json','--archive','/home/ubuntu/joxing/downloads/LocomotionFull-2b2f0f10.zip','--out','runs/v0.0.8/physical-01'],remaining())
 run('audit',['uv','run','--project','experiments/v0.0.8/runtime','--locked','python','scripts/verify_physical_motor.py','--run','runs/v0.0.8/physical-01','--out','runs/v0.0.8/audit-linux.json'],remaining())
 run('report',['uv','run','--project','experiments/v0.0.8/runtime','--locked','python','scripts/report_physical_motor.py','--run','runs/v0.0.8/physical-01','--audit','runs/v0.0.8/audit-linux.json','--out','runs/v0.0.8/report','--video'],remaining())
 status['status']='completed'
except Exception as error:
 status.update(status='failed',error=str(error),traceback=traceback.format_exc())
finally:
 status['finished_utc']=datetime.datetime.now(datetime.timezone.utc).isoformat();save()
